# Desktop build of the audio core tests and the speech-timing test (CI and local).
cmake_minimum_required(VERSION 3.22)
project(lectern_tests C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CPP ${CMAKE_CURRENT_SOURCE_DIR}/../cpp)
find_package(Threads REQUIRED)

add_executable(engine_test engine_test.cpp ${CPP}/core.cpp)
target_include_directories(engine_test PRIVATE ${CPP})
target_link_libraries(engine_test Threads::Threads)

option(LECTERN_ASR_TEST "build the speech-timing test (needs whisper.cpp)" ON)
if(LECTERN_ASR_TEST)
  include(FetchContent)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
  if(DEFINED ENV{WHISPER_SRC})
    FetchContent_Declare(whisper SOURCE_DIR $ENV{WHISPER_SRC})
  else()
    FetchContent_Declare(whisper GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git GIT_TAG v1.8.2 GIT_SHALLOW TRUE)
  endif()
  FetchContent_MakeAvailable(whisper)
  add_executable(asr_test asr_test.cpp ${CPP}/core.cpp ${CPP}/transcribe.cpp)
  target_include_directories(asr_test PRIVATE ${CPP})
  target_link_libraries(asr_test whisper Threads::Threads)
endif()
